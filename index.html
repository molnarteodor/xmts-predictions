function analyzeMatch(ev, lg) { const comp = ev.competitions[0], cs = comp.competitors; const home = cs.find(c => c.homeAway === "home") || cs[0], away = cs.find(c => c.homeAway === "away") || cs[1]; const hName = home.team.displayName || home.team.name, aName = away.team.displayName || away.team.name; const dt = new Date(comp.date || ev.date), stState = (comp.status && comp.status.type && comp.status.type.state) || "pre"; const hScore = home.score !== undefined ? parseInt(home.score, 10) : null, aScore = away.score !== undefined ? parseInt(away.score, 10) : null; let seed = 0; for (let i = 0; i < (hName + aName).length; i++) seed = (seed * 37 + (hName + aName).charCodeAt(i)) % 10000; const hInfo = extractTeamStrength(home, seed); const aInfo = extractTeamStrength(away, seed + 7); const hBoost = getEliteBoost(hName) + hInfo.score; const aBoost = getEliteBoost(aName) + aInfo.score; // Bază xG calibrată pe media ligii (fără umflare artificială) const leagueGoalFactor = ((lg.cBase || 9.5) - 9.5) * 0.12; const lamH = Math.max(0.55, Math.min(3.10, 1.32 + hBoost * 1.15 - aBoost * 0.45 + leagueGoalFactor)); const lamA = Math.max(0.45, Math.min(2.70, 1.02 + aBoost * 1.15 - hBoost * 0.45 + leagueGoalFactor * 0.8)); let pH = 0, pD = 0, pA = 0, pO15 = 0, pO25 = 0, pGG = 0; for (let h = 0; h <= 7; h++) for (let a = 0; a <= 7; a++) { const p = poissonPmf(h, lamH) * poissonPmf(a, lamA); if (h > a) pH += p; else if (h === a) pD += p; else pA += p; if (h + a > 1.5) pO15 += p; if (h + a > 2.5) pO25 += p; if (h > 0 && a > 0) pGG += p; } const tot = pH + pD + pA; pH /= tot; pD /= tot; pA /= tot; const p1X = pH + pD, pX2 = pA + pD; // Calculăm cotele reale (fără Math.max(1.20) care bloca totul pe Peste 1.5 Goluri) const candidates = [ { name: `1 (${hName})`, prob: pH, odd: +(0.93 / Math.max(0.15, pH)).toFixed(2), w: 1.12 }, { name: `2 (${aName})`, prob: pA, odd: +(0.93 / Math.max(0.15, pA)).toFixed(2), w: 1.12 }, { name: "Peste 2.5 Goluri", prob: pO25, odd: +(0.93 / Math.max(0.15, pO25)).toFixed(2), w: 1.10 }, { name: "GG (Ambele marchează)", prob: pGG, odd: +(0.93 / Math.max(0.15, pGG)).toFixed(2), w: 1.08 }, { name: "Șansă Dublă 1X", prob: p1X, odd: +(0.91 / Math.max(0.20, p1X)).toFixed(2), w: 0.96 }, { name: "Șansă Dublă X2", prob: pX2, odd: +(0.91 / Math.max(0.20, pX2)).toFixed(2), w: 0.96 }, { name: "Peste 1.5 Goluri", prob: pO15, odd: +(0.91 / Math.max(0.20, pO15)).toFixed(2), w: 0.94 } ]; candidates.forEach(c => { // Zona optimă de Value Bet este între cotele 1.36 și 2.15 let zoneMultiplier = 1.0; if (c.odd >= 1.38 && c.odd <= 2.10) zoneMultiplier = 1.25; else if (c.odd >= 1.28 && c.odd < 1.38) zoneMultiplier = 1.05; else if (c.odd < 1.28) zoneMultiplier = 0.65; // Penalizăm cotele de 1.15-1.25 c.valueScore = c.prob * c.w * zoneMultiplier; }); // Filtrăm doar opțiunile cu cotă reală >= 1.30 și șanse reale >= 52% let valid = candidates.filter(c => c.odd >= 1.30 && c.odd <= 2.45 && c.prob >= 0.52); if (!valid.length) valid = candidates.filter(c => c.odd >= 1.22 && c.prob >= 0.50); if (!valid.length) valid = candidates; valid.sort((a, b) => b.valueScore - a.valueScore); const best = valid[0]; // Extragere statistici reale din ESPN (dacă meciul e LIVE sau FT) const getStat = (compObj, names) => { if (!compObj.statistics) return null; for (let n of names) { const f = compObj.statistics.find(s => s.name === n || s.abbreviation === n); if (f && f.displayValue !== undefined) return parseFloat(f.displayValue); } return null; }; const realHCorners = getStat(home, ["cornerKicks", "corners", "CK"]); const realACorners = getStat(away, ["cornerKicks", "corners", "CK"]); const realHYellow = getStat(home, ["yellowCards", "YC"]); const realAYellow = getStat(away, ["yellowCards", "YC"]); const realHRed = getStat(home, ["redCards", "RC"]) || 0; const realARed = getStat(away, ["redCards", "RC"]) || 0; // Calcul matematic pentru Cornere (bazat pe volumul ofensiv xG și media ligii, fără seed aleatoriu) const totalXG = lamH + lamA; const dominancePush = Math.abs(lamH - lamA) * 0.45; const expCorners = Math.max(7.2, Math.min(12.2, +((lg.cBase || 9.5) + (totalXG - 2.35) * 1.35 + dominancePush).toFixed(1))); const cornerLine = expCorners >= 10.2 ? "Peste 9.5" : (expCorners >= 9.1 ? "Peste 8.5" : "Sub 10.5"); // Calcul matematic pentru Cartonașe (bazat pe intensitatea/echilibrul meciului și media ligii) const closeness = 1 - Math.min(1, Math.abs(pH - pA) * 1.6); // 1 = meci foarte strâns, 0 = dezechilibrat const expCards = Math.max(3.0, Math.min(6.2, +((lg.kBase || 4.3) + (closeness - 0.5) * 1.1).toFixed(1))); const cardLine = expCards >= 4.8 ? "Peste 4.5" : (expCards >= 3.9 ? "Peste 3.5" : "Sub 4.5"); const actualCorners = (realHCorners !== null && realACorners !== null) ? (realHCorners + realACorners) : null; const actualCards = (realHYellow !== null && realAYellow !== null) ? (realHYellow + realAYellow + realHRed + realARed) : null; return { id: ev.id, league: lg.name, icon: lg.icon, cat: lg.cat, date: roDateStr(dt), time: roTimeStr(dt), status: stState === "post" ? "FT" : (stState === "in" ? "LIVE" : "NS"), home: hName, away: aName, hScore, aScore, hForm: hInfo.formDisplay, aForm: aInfo.formDisplay, lamH: lamH.toFixed(2), lamA: lamA.toFixed(2), pick: best.name, prob: Math.round(best.prob * 100), odd: +best.odd.toFixed(2), tipsterScore: best.valueScore, expCorners, cornerLine, actualCorners, expCards, cardLine, actualCards, probs: { p1: Math.round(pH*100), pX: Math.round(pD*100), p2: Math.round(pA*100), o25: Math.round(pO25*100), gg: Math.round(pGG*100), o15: Math.round(pO15*100) } }; } function renderMatchCard(m) { const stClass = m.status === "FT" ? "st-ft" : (m.status === "LIVE" ? "st-live" : "st-ns"); let cardStatusClass = ""; if (m.status === "FT" && m.hScore !== null && m.aScore !== null) { const outcome = checkBetOutcome(m.pick, m.hScore, m.aScore); if (outcome === true) cardStatusClass = "match-win"; else if (outcome === false) cardStatusClass = "match-loss"; } const stText = m.status === "FT" ? `FT (${m.hScore}-${m.aScore})` : (m.status === "LIVE" ? `LIVE ${m.hScore}-${m.aScore}` : `🕒 ${m.time}`); const center = (m.status === "FT" || m.status === "LIVE") ? `${m.hScore} - ${m.aScore}` : "VS"; const joeBtnHtml = hasJoeAccess() ? `` : ""; const cornersDisplay = m.actualCorners !== null ? `${m.actualCorners} (Est: ${m.expCorners} · ${m.cornerLine})` : `${m.cornerLine} (~${m.expCorners})`; const cardsDisplay = m.actualCards !== null ? `${m.actualCards} (Est: ${m.expCards} · ${m.cardLine})` : `${m.cardLine} (~${m.expCards})`; return `
${m.icon} ${m.league} · ${m.date}${stText}
${m.home}
${center}
${m.away}
⭐ ${m.pick}
Siguranță: ${m.prob}% · xG: ${m.lamH}-${m.lamA}
@${m.odd.toFixed(2)}
${joeBtnHtml}
1${m.probs.p1}%
X${m.probs.pX}%
2${m.probs.p2}%
Peste 1.5${m.probs.o15}%
Peste 2.5${m.probs.o25}%
GG${m.probs.gg}%
📈 Formă: ${m.hForm} vs ${m.aForm}
🚩 Cornere: ${cornersDisplay} | 🟨 Cartonașe: ${cardsDisplay}
`; }
